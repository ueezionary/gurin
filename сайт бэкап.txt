    const container = document.querySelector('.presentation-container');

    const dots = document.querySelectorAll('.pagination-dot');

    const links = document.querySelectorAll('.nav-links a');

    const slides = document.querySelectorAll('.slide');



    let currentSlide = 0;

    let isThrottled = false;

    let lastInternalScrollTime = 0;



    // 🌐 LANGUAGE BUTTONS HANDLER

    const langBtns = document.querySelectorAll('.lang-btn');

    langBtns.forEach(btn => {

      btn.addEventListener('click', () => {

        langBtns.forEach(b => b.classList.remove('active'));

        btn.classList.add('active');

        const lang = btn.getAttribute('data-lang');

        document.body.className = `lang-${lang}`; 

      });

    });



    // ПЕРЕХВАТ КЛИКОВ НАВИГАЦИИ ДЛЯ ПЛАВНОГО ДВИЖЕНИЯ СНЭПА

    const allNavAnchors = document.querySelectorAll('.nav-logo, .nav-links a, .pagination-dot');

    allNavAnchors.forEach(anchor => {

      anchor.addEventListener('click', (e) => {

        e.preventDefault(); 

        

        const targetId = anchor.getAttribute('href');

        const targetSlide = document.querySelector(targetId);

        

        slides.forEach((slide, index) => {

          if (slide.getAttribute('id') === targetId.replace('#', '')) {

            currentSlide = index;

          }

        });



        targetSlide.scrollIntoView({ 

          behavior: 'smooth', 

          block: 'nearest', 

          inline: 'start' 

        });

      });

    });



    // AUTOMATED DYNAMIC INTERCEPTOR FOR SCROLLABLE VIEWPORTS WITH 350MS TIME-LOCK

    container.addEventListener('wheel', (e) => {

      if (e.deltaY === 0) return;

      e.preventDefault(); 



      if (isThrottled) return;



      const activeSlide = slides[currentSlide];

      const scrollBox = activeSlide.querySelector('.works-scroll-box, .gallery-grid');



      if (scrollBox) {

        const isAtBottom = scrollBox.scrollTop + scrollBox.clientHeight >= scrollBox.scrollHeight - 5;

        const isAtTop = scrollBox.scrollTop <= 5;



        if (e.deltaY > 0) {

          if (!isAtBottom) {

            scrollBox.scrollTop += e.deltaY;

            lastInternalScrollTime = Date.now();

            return;

          }

          if (Date.now() - lastInternalScrollTime < 350) return; 

        } else {

          if (!isAtTop) {

            scrollBox.scrollTop += e.deltaY;

            lastInternalScrollTime = Date.now();

            return;

          }

          if (Date.now() - lastInternalScrollTime < 350) return; 

        }

      }



      if (e.deltaY > 0) {

        if (currentSlide < slides.length - 1) currentSlide++;

      } else {

        if (currentSlide > 0) currentSlide--;

      }



      isThrottled = true;



      slides[currentSlide].scrollIntoView({ 

        behavior: 'smooth', 

        block: 'nearest', 

        inline: 'start' 

      });



      setTimeout(() => {

        isThrottled = false;

      }, 600);

    }, { passive: false });



    // NAVIGATION LINK SYNC AND OBSERVER LAYER

    const observerOptions = {

      root: container,

      threshold: 0.5

    };



    const observerCallback = (entries) => {

      entries.forEach(entry => {

        if (entry.isIntersecting) {

          dots.forEach(dot => dot.classList.remove('active'));

          links.forEach(link => link.classList.remove('active'));

          

          const id = entry.target.getAttribute('id');

          

          slides.forEach((slide, index) => {

            if(slide.getAttribute('id') === id) currentSlide = index;

          });



          const currentDot = document.querySelector(`.pagination-dot[href="#${id}"]`);

          if (currentDot) currentDot.classList.add('active');



          const currentLink = document.querySelector(`.nav-links a[href="#${id}"]`);

          if (currentLink) currentLink.classList.add('active');

        }

      });

    };



    const observer = new IntersectionObserver(observerCallback, observerOptions);

    slides.forEach(slide => observer.observe(slide));
